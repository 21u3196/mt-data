<?php

include("../config.php");

if(!isset($_SESSION['user_id']))
{
    header("Location: login.php");
    exit();
}

$user_id = $_SESSION['user_id'];

if($_SERVER['REQUEST_METHOD'] != 'POST')
{
    header("Location: dashboard.php");
    exit();
}

$network = mysqli_real_escape_string(
    $conn,
    $_POST['network']
);

$phone = mysqli_real_escape_string(
    $conn,
    $_POST['phone']
);

$plan_id = intval(
    $_POST['plan']
);

/* ==========================
   GET USER DETAILS
========================== */

$user_query = mysqli_query(
    $conn,
    "SELECT * FROM users
     WHERE id='$user_id'"
);

$user = mysqli_fetch_assoc(
    $user_query
);

/* ==========================
   GET PLAN DETAILS
========================== */

$plan_query = mysqli_query(
    $conn,
    "SELECT
        data_plans.*,
        networks.network_name
     FROM data_plans
     INNER JOIN networks
     ON data_plans.network_id = networks.id
     WHERE data_plans.id='$plan_id'"
);

if(mysqli_num_rows($plan_query) == 0)
{
    die("Invalid Data Plan");
}

$plan = mysqli_fetch_assoc(
    $plan_query
);

$amount = $plan['amount'];

$current_balance =
$user['wallet_balance'];

/* ==========================
   CHECK WALLET BALANCE
========================== */

if($current_balance < $amount)
{
?>

<!DOCTYPE html>
<html>
<head>

<title>Insufficient Balance</title>

<link rel="stylesheet"
href="../assets/css/style.css">

</head>

<body>

<div class="alert-danger">

Insufficient Wallet Balance

</div>

<br>

<a href="dashboard.php">

<button class="btn btn-danger">

Back To Dashboard

</button>

</a>

</div>

</div>

</body>

</html>

<?php
exit();
}

/* ==========================
   DEDUCT BALANCE
========================== */

$new_balance =
$current_balance - $amount;

mysqli_query(
$conn,
"UPDATE users
SET wallet_balance='$new_balance'
WHERE id='$user_id'"
);

/* ==========================
   SAVE TRANSACTION
========================== */

$description =
$plan['network_name']
.
" "
.
$plan['plan_name']
.
" Data Purchase";

mysqli_query(
$conn,
"INSERT INTO transactions(
user_id,
service_type,
description,
phone_number,
amount
)
VALUES(
'$user_id',
'Data',
'$description',
'$phone',
'$amount'
)"
);
