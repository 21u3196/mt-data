<?php

include("../config.php");

if(!isset($_SESSION['user_id']))
{
    header("Location: login.php");
    exit();
}

$user_id = $_SESSION['user_id'];

if($_SERVER['REQUEST_METHOD'] != "POST")
{
    header("Location: dashboard.php");
    exit();
}

$network = mysqli_real_escape_string(
    $conn,
    $_POST['network']
);

$phone = mysqli_real_escape_string(
    $conn,
    $_POST['phone']
);

$amount = floatval(
    $_POST['amount']
);

if($amount <= 0)
{
    die("Invalid Amount");
}

/* ======================
   USER INFORMATION
====================== */

$user_query = mysqli_query(
$conn,
"SELECT * FROM users
WHERE id='$user_id'"
);

$user =
mysqli_fetch_assoc(
$user_query
);

$current_balance =
$user['wallet_balance'];

/* ======================
   CHECK BALANCE
====================== */

if($current_balance < $amount)
{
?>

<!DOCTYPE html>
<html>

<head>

<title>
Insufficient Balance
</title>

<link rel="stylesheet"
href="../assets/css/style.css">

</head>

<body>

<div class="container">

<div class="card">

<div class="alert-danger">

Insufficient Wallet Balance

</div>

<br>

<a href="dashboard.php">

<button class="btn btn-danger">

Back To Dashboard

</button>

</a>

</div>

</div>

</body>

</html>

<?php
exit();
}

/* ======================
   DEDUCT BALANCE
====================== */

$new_balance =
$current_balance - $amount;

mysqli_query(
$conn,
"UPDATE users
SET wallet_balance='$new_balance'
WHERE id='$user_id'"
);

/* ======================
   SAVE TRANSACTION
====================== */

$description =
$network .
" Airtime Purchase";

mysqli_query(
$conn,
"INSERT INTO transactions(
user_id,
service_type,
description,
phone_number,
amount
)
VALUES(
'$user_id',
'Airtime',
'$description',
'$phone',
'$amount'
)"
);

?>

<!DOCTYPE html>
<html>

<head>

<title>

Airtime Purchase Successful

</title>

<link rel="stylesheet"
href="../assets/css/style.css">

</head>

<body>

<div class="container">

<div class="card">

<div class="alert-success">

Airtime Purchase Successful

</div>

<br>

<h3>

Network:
<?php echo $network; ?>

</h3>

<h3>

Phone:
<?php echo $phone; ?>

</h3>

<h3>

Amount:
₦<?php echo number_format($amount,2); ?>

</h3>

<h3>

Remaining Wallet:
₦<?php echo number_format($new_balance,2); ?>

</h3>

<br>

<a href="dashboard.php">

<button class="btn btn-primary">

Return Dashboard

</button>

</a>

</div>

</div>

</body>

</html>
